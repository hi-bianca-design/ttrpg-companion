# Auth Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the full auth module — Discord + Apple OAuth, username-setup hard gate, email verification soft gate, OTP password reset, profile screen with avatar, and account deletion.

**Architecture:** Supabase Auth handles all credential flows (OAuth tokens, OTP emails, session refresh). A `public.users` profile row is created after every sign-up to store username, display_name, and avatar_url. The root layout gates navigation on both session presence AND profile existence, directing new users to a username-setup screen before the app.

**Tech Stack:** Supabase Auth (email/password + OAuth), expo-router, expo-image-picker, NativeWind, Zustand (not used here — plain React state is sufficient).

```typescript
// src/modules/auth/auth.service.ts
import { supabase } from '@/lib/supabase'

// ─── Email / Password ───────────────────────────────────────────────────────

export async function signUp(email: string, password: string) {
  const { data, error } = await supabase.auth.signUp({ email, password })
  if (error) throw new Error(error.message)
  return data
}

export async function signIn(email: string, password: string) {
  const { data, error } = await supabase.auth.signInWithPassword({ email, password })
  if (error) throw new Error(error.message)
  return data
}

export async function signOut() {
  const { error } = await supabase.auth.signOut()
  if (error) throw new Error(error.message)
}

// ─── OAuth ───────────────────────────────────────────────────────────────────

export async function signInWithDiscord() {
  const { data, error } = await supabase.auth.signInWithOAuth({ provider: 'discord' })
  if (error) throw new Error(error.message)
  return data
}

export async function signInWithApple() {
  const { data, error } = await supabase.auth.signInWithOAuth({ provider: 'apple' })
  if (error) throw new Error(error.message)
  return data
}

// ─── Password Reset (OTP) ────────────────────────────────────────────────────

export async function requestPasswordReset(email: string) {
  const { error } = await supabase.auth.signInWithOtp({
    email,
    options: { shouldCreateUser: false },
  })
  if (error) throw new Error(error.message)
}

export async function verifyOTP(email: string, token: string) {
  const { data, error } = await supabase.auth.verifyOtp({ email, token, type: 'email' })
  if (error) throw new Error(error.message)
  return data
}

export async function resetPassword(newPassword: string) {
  const { error } = await supabase.auth.updateUser({ password: newPassword })
  if (error) throw new Error(error.message)
}

// ─── Profile ─────────────────────────────────────────────────────────────────

export async function checkUsernameAvailable(username: string): Promise<boolean> {
  const { data } = await supabase
    .from('users')
    .select('id')
    .eq('username', username)
    .maybeSingle()
  return data === null
}

export async function setupProfile({
  id,
  email,
  username,
  display_name,
}: {
  id: string
  email: string
  username: string
  display_name?: string
}) {
  const { error } = await supabase.from('users').insert({ id, email, username, display_name })
  if (error) throw new Error(error.message)
}

export async function getProfile(userId: string) {
  const { data, error } = await supabase
    .from('users')
    .select('id, email, username, display_name, avatar_url')
    .eq('id', userId)
    .single()
  if (error) throw new Error(error.message)
  return data
}

export async function updateProfile(
  userId: string,
  updates: { username?: string; display_name?: string; avatar_url?: string }
) {
  const { error } = await supabase.from('users').update(updates).eq('id', userId)
  if (error) throw new Error(error.message)
}

export async function deleteAccount(userId: string) {
  // Delete profile row first (cascade handles related data)
  const { error: profileError } = await supabase.from('users').delete().eq('id', userId)
  if (profileError) throw new Error(profileError.message)
  // Sign out locally; server-side user deletion requires a Supabase Edge Function or admin key
  await signOut()
}

// ─── Sign all devices out ─────────────────────────────────────────────────────

export async function signOutAllDevices() {
  const { error } = await supabase.auth.signOut({ scope: 'global' })
  if (error) throw new Error(error.message)
}
```

```typescript
// src/modules/auth/AuthProvider.tsx
import React, { createContext, useEffect, useState, useCallback } from 'react'
import { Session, User } from '@supabase/supabase-js'
import { supabase } from '@/lib/supabase'
import { getProfile } from './auth.service'

export type UserProfile = {
  id: string
  email: string
  username: string
  display_name: string | null
  avatar_url: string | null
}

type AuthContextValue = {
  user: User | null
  session: Session | null
  profile: UserProfile | null
  loading: boolean
  refreshProfile: () => Promise<void>
}

export const AuthContext = createContext<AuthContextValue | null>(null)

export function AuthProvider({ children }: { children: React.ReactNode }) {
  const [user, setUser] = useState<User | null>(null)
  const [session, setSession] = useState<Session | null>(null)
  const [profile, setProfile] = useState<UserProfile | null>(null)
  const [loading, setLoading] = useState(true)

  const loadProfile = useCallback(async (userId: string) => {
    try {
      const p = await getProfile(userId)
      setProfile(p)
    } catch {
      setProfile(null)
    }
  }, [])

  const refreshProfile = useCallback(async () => {
    if (user) await loadProfile(user.id)
  }, [user, loadProfile])

  useEffect(() => {
    supabase.auth.getSession()
      .then(async ({ data: { session } }) => {
        setSession(session)
        setUser(session?.user ?? null)
        if (session?.user) await loadProfile(session.user.id)
      })
      .catch(() => {})
      .finally(() => setLoading(false))

    const { data: { subscription } } = supabase.auth.onAuthStateChange(async (event, session) => {
      if (event === 'INITIAL_SESSION') return
      setSession(session)
      setUser(session?.user ?? null)
      if (session?.user) {
        await loadProfile(session.user.id)
      } else {
        setProfile(null)
      }
    })

    return () => subscription.unsubscribe()
  }, [loadProfile])

  return (
    <AuthContext.Provider value={{ user, session, profile, loading, refreshProfile }}>
      {children}
    </AuthContext.Provider>
  )
}
```

```typescript
// app/_layout.tsx
import '../global.css'
import { useEffect } from 'react'
import { Slot, useRouter, useSegments } from 'expo-router'
import { AuthProvider } from '@/modules/auth/AuthProvider'
import { useAuth } from '@/modules/auth/useAuth'

function RootRedirect() {
  const { user, profile, loading } = useAuth()
  const router = useRouter()
  const segments = useSegments()

  useEffect(() => {
    if (loading) return
    const inAuthGroup = segments[0] === '(auth)'
    const inUsernameSetup = segments[1] === 'username-setup'

    if (!user && !inAuthGroup) {
      router.replace('/(auth)/login')
      return
    }

    if (user && !profile && !inUsernameSetup) {
      // Authenticated but no profile row — needs username setup
      router.replace('/(auth)/username-setup')
      return
    }

    if (user && profile && inAuthGroup) {
      router.replace('/(app)')
    }
  }, [user, profile, loading, segments])

  return <Slot />
}

export default function RootLayout() {
  return (
    <AuthProvider>
      <RootRedirect />
    </AuthProvider>
  )
}
```

```typescript
// app/(auth)/username-setup.tsx
import { useState } from 'react'
import { View, Text, TextInput, TouchableOpacity, ActivityIndicator } from 'react-native'
import { useAuth } from '@/modules/auth/useAuth'
import { setupProfile, checkUsernameAvailable } from '@/modules/auth/auth.service'

export default function UsernameSetupScreen() {
  const { user, refreshProfile } = useAuth()
  const [username, setUsername] = useState('')
  const [displayName, setDisplayName] = useState('')
  const [usernameStatus, setUsernameStatus] = useState<'idle' | 'checking' | 'available' | 'taken'>('idle')
  const [loading, setLoading] = useState(false)
  const [error, setError] = useState<string | null>(null)

  async function handleUsernameBlur() {
    const trimmed = username.trim().toLowerCase()
    if (!trimmed || trimmed.length < 3) return
    setUsernameStatus('checking')
    const available = await checkUsernameAvailable(trimmed)
    setUsernameStatus(available ? 'available' : 'taken')
  }

  async function handleSubmit() {
    if (!user) return
    const trimmedUsername = username.trim().toLowerCase()
    if (!trimmedUsername || trimmedUsername.length < 3) {
      setError('Username must be at least 3 characters.')
      return
    }
    if (usernameStatus === 'taken') {
      setError('That username is already taken.')
      return
    }
    setLoading(true)
    setError(null)
    try {
      await setupProfile({
        id: user.id,
        email: user.email ?? '',
        username: trimmedUsername,
        display_name: displayName.trim() || null,
      })
      await refreshProfile()
      // Root layout will redirect to /(app) once profile is set
    } catch (e: unknown) {
      setError(e instanceof Error ? e.message : 'Something went wrong.')
    } finally {
      setLoading(false)
    }
  }

  const usernameHint =
    usernameStatus === 'checking' ? 'Checking...' :
    usernameStatus === 'available' ? '✓ Available' :
    usernameStatus === 'taken' ? '✗ Already taken' :
    'Used for campaign invites. Min 3 characters.'

  const usernameHintColor =
    usernameStatus === 'available' ? 'text-green-400' :
    usernameStatus === 'taken' ? 'text-red-400' :
    'text-muted'

  return (
    <View className="flex-1 justify-center px-6">
      <Text className="text-4xl font-bold text-primary mb-2">One more step</Text>
      <Text className="text-muted mb-10">Set up your identity</Text>

      <TextInput
        className="bg-surface text-text rounded-xl px-4 py-3 mb-1"
        placeholder="Username"
        placeholderTextColor="#6c7086"
        value={username}
        onChangeText={(v) => { setUsername(v); setUsernameStatus('idle') }}
        onBlur={handleUsernameBlur}
        autoCapitalize="none"
      />
      <Text className={`text-sm mb-4 ml-1 ${usernameHintColor}`}>{usernameHint}</Text>

      <TextInput
        className="bg-surface text-text rounded-xl px-4 py-3 mb-6"
        placeholder="Display name (optional)"
        placeholderTextColor="#6c7086"
        value={displayName}
        onChangeText={setDisplayName}
      />

      {error && <Text className="text-red-400 mb-4 text-center">{error}</Text>}

      <TouchableOpacity
        className="bg-primary rounded-xl py-3 items-center"
        onPress={handleSubmit}
        disabled={loading || usernameStatus === 'taken' || usernameStatus === 'checking'}
      >
        {loading
          ? <ActivityIndicator color="#1e1e2e" />
          : <Text className="text-base font-semibold text-[#1e1e2e]">Continue</Text>
        }
      </TouchableOpacity>
    </View>
  )
}
```

```typescript
// app/(auth)/otp-verify.tsx
import { useState, useRef } from 'react'
import { View, Text, TextInput, TouchableOpacity, ActivityIndicator } from 'react-native'
import { useRouter, useLocalSearchParams } from 'expo-router'
import { verifyOTP, requestPasswordReset } from '@/modules/auth/auth.service'

export default function OTPVerifyScreen() {
  const { email } = useLocalSearchParams<{ email: string }>()
  const [code, setCode] = useState('')
  const [loading, setLoading] = useState(false)
  const [resendCooldown, setResendCooldown] = useState(0)
  const [error, setError] = useState<string | null>(null)
  const router = useRouter()
  const cooldownRef = useRef<ReturnType<typeof setInterval> | null>(null)

  async function handleVerify() {
    if (code.length !== 6) return
    setLoading(true)
    setError(null)
    try {
      await verifyOTP(email, code)
      router.replace('/(auth)/reset-password')
    } catch (e: unknown) {
      setError(e instanceof Error ? e.message : 'Invalid or expired code.')
    } finally {
      setLoading(false)
    }
  }

  async function handleResend() {
    if (resendCooldown > 0) return
    await requestPasswordReset(email)
    setResendCooldown(60)
    cooldownRef.current = setInterval(() => {
      setResendCooldown((s) => {
        if (s <= 1) { clearInterval(cooldownRef.current!); return 0 }
        return s - 1
      })
    }, 1000)
  }

  return (
    <View className="flex-1 justify-center px-6">
      <Text className="text-4xl font-bold text-primary mb-2">Enter Code</Text>
      <Text className="text-muted mb-10">We sent a 6-digit code to {email}</Text>

      <TextInput
        className="bg-surface text-text rounded-xl px-4 py-3 mb-2 text-center text-2xl tracking-widest"
        placeholder="------"
        placeholderTextColor="#6c7086"
        value={code}
        onChangeText={(v) => setCode(v.replace(/\D/g, '').slice(0, 6))}
        keyboardType="number-pad"
        maxLength={6}
      />

      <TouchableOpacity className="mb-6 self-center" onPress={handleResend} disabled={resendCooldown > 0}>
        <Text className="text-primary text-sm">
          {resendCooldown > 0 ? `Resend in ${resendCooldown}s` : 'Resend code'}
        </Text>
      </TouchableOpacity>

      {error && <Text className="text-red-400 mb-4 text-center">{error}</Text>}

      <TouchableOpacity
        className="bg-primary rounded-xl py-3 items-center"
        onPress={handleVerify}
        disabled={loading || code.length !== 6}
      >
        {loading
          ? <ActivityIndicator color="#1e1e2e" />
          : <Text className="text-base font-semibold text-[#1e1e2e]">Verify</Text>
        }
      </TouchableOpacity>
```

```typescript
// app/(app)/profile/index.tsx
import { useState } from 'react'
import { View, Text, TextInput, TouchableOpacity, Image, ActivityIndicator, Alert, ScrollView } from 'react-native'
import * as ImagePicker from 'expo-image-picker'
import { useAuth } from '@/modules/auth/useAuth'
import { updateProfile, deleteAccount, signOut, signOutAllDevices, checkUsernameAvailable } from '@/modules/auth/auth.service'
import { supabase } from '@/lib/supabase'

export default function ProfileScreen() {
  const { user, profile, refreshProfile } = useAuth()
  const [username, setUsername] = useState(profile?.username ?? '')
  const [displayName, setDisplayName] = useState(profile?.display_name ?? '')
  const [avatarUri, setAvatarUri] = useState<string | null>(profile?.avatar_url ?? null)
  const [usernameStatus, setUsernameStatus] = useState<'idle' | 'checking' | 'available' | 'taken'>('idle')
  const [saving, setSaving] = useState(false)
  const [error, setError] = useState<string | null>(null)

  async function handlePickAvatar() {
    const result = await ImagePicker.launchImageLibraryAsync({
      mediaTypes: ImagePicker.MediaTypeOptions.Images,
      allowsEditing: true,
      aspect: [1, 1],
      quality: 0.8,
    })
    if (!result.canceled && result.assets[0]) {
      setAvatarUri(result.assets[0].uri)
    }
  }

  async function handleUsernameBlur() {
    const trimmed = username.trim().toLowerCase()
    if (trimmed === profile?.username) { setUsernameStatus('idle'); return }
    if (trimmed.length < 3) return
    setUsernameStatus('checking')
    const available = await checkUsernameAvailable(trimmed)
    setUsernameStatus(available ? 'available' : 'taken')
  }

  async function handleSave() {
    if (!user || !profile) return
    if (usernameStatus === 'taken') { setError('That username is already taken.'); return }
    setSaving(true)
    setError(null)
    try {
      let avatar_url = profile.avatar_url

      if (avatarUri && avatarUri !== profile.avatar_url) {
        // Upload avatar to Supabase Storage
        const ext = avatarUri.split('.').pop() ?? 'jpg'
        const path = `avatars/${user.id}.${ext}`
        const response = await fetch(avatarUri)
        const blob = await response.blob()
        const { error: uploadError } = await supabase.storage
          .from('avatars')
          .upload(path, blob, { upsert: true, contentType: `image/${ext}` })
        if (uploadError) throw new Error(uploadError.message)
        const { data: { publicUrl } } = supabase.storage.from('avatars').getPublicUrl(path)
        avatar_url = publicUrl
      }

      await updateProfile(user.id, {
        username: username.trim().toLowerCase(),
        display_name: displayName.trim() || null,
        avatar_url,
      })
      await refreshProfile()
    } catch (e: unknown) {
      setError(e instanceof Error ? e.message : 'Save failed.')
    } finally {
      setSaving(false)
    }
  }

  function handleDeleteAccount() {
    Alert.alert(
      'Delete Account',
      "Your account and characters will be deleted. Campaigns you own will remain accessible to your players with your name shown as '[Deleted User]'. Type your username to confirm.",
      [
        { text: 'Cancel', style: 'cancel' },
        {
          text: 'Delete',
          style: 'destructive',
          onPress: () => {
            Alert.prompt(
              'Confirm',
              `Type "${profile?.username}" to delete your account.`,
              async (input) => {
                if (input !== profile?.username) {
                  Alert.alert('Incorrect username', 'Account not deleted.')
                  return
                }
                if (user) await deleteAccount(user.id)
              }
            )
          },
        },
      ]
    )
  }

  return (
    <ScrollView className="flex-1 px-6 pt-12">
      <Text className="text-3xl font-bold text-primary mb-8">Profile</Text>

      {/* Avatar */}
      <TouchableOpacity className="items-center mb-6" onPress={handlePickAvatar}>
        {avatarUri ? (
          <Image source={{ uri: avatarUri }} className="w-24 h-24 rounded-full" />
        ) : (
          <View className="w-24 h-24 rounded-full bg-surface items-center justify-center">
            <Text className="text-3xl text-muted">{profile?.username?.[0]?.toUpperCase() ?? '?'}</Text>
          </View>
        )}
        <Text className="text-primary text-sm mt-2">Change photo</Text>
      </TouchableOpacity>

      {/* Username */}
      <Text className="text-muted text-sm mb-1 ml-1">Username</Text>
      <TextInput
        className="bg-surface text-text rounded-xl px-4 py-3 mb-1"
        value={username}
        onChangeText={(v) => { setUsername(v); setUsernameStatus('idle') }}
        onBlur={handleUsernameBlur}
        autoCapitalize="none"
      />
      {usernameStatus !== 'idle' && (
        <Text className={`text-sm mb-3 ml-1 ${usernameStatus === 'available' ? 'text-green-400' : usernameStatus === 'taken' ? 'text-red-400' : 'text-muted'}`}>
          {usernameStatus === 'checking' ? 'Checking...' : usernameStatus === 'available' ? '✓ Available' : '✗ Already taken'}
        </Text>
      )}

      {/* Display Name */}
      <Text className="text-muted text-sm mb-1 ml-1 mt-2">Display Name</Text>
      <TextInput
        className="bg-surface text-text rounded-xl px-4 py-3 mb-6"
        value={displayName}
        onChangeText={setDisplayName}
        placeholder="Optional"
        placeholderTextColor="#6c7086"
      />

      {error && <Text className="text-red-400 mb-4 text-center">{error}</Text>}

      <TouchableOpacity
        className="bg-primary rounded-xl py-3 items-center mb-4"
        onPress={handleSave}
        disabled={saving}
      >
        {saving
          ? <ActivityIndicator color="#1e1e2e" />
          : <Text className="text-base font-semibold text-[#1e1e2e]">Save Changes</Text>
        }
      </TouchableOpacity>

      <TouchableOpacity
        className="bg-surface rounded-xl py-3 items-center mb-3"
        onPress={signOut}
      >
        <Text className="text-text font-semibold">Sign Out</Text>
      </TouchableOpacity>

      <TouchableOpacity
        className="bg-surface rounded-xl py-3 items-center mb-10"
        onPress={signOutAllDevices}
      >
        <Text className="text-text font-semibold">Sign Out All Devices</Text>
      </TouchableOpacity>

      <TouchableOpacity className="items-center mb-16" onPress={handleDeleteAccount}>
        <Text className="text-red-400 text-sm">Delete Account</Text>
      </TouchableOpacity>
    </ScrollView>
  )
}
```
